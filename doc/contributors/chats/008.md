```ruby
# In App#dispatch_ipc
def dispatch_ipc(event)
  handler = @builder.invoke_handler_hook
  invoke = Invoke.new(command: event.command, payload: event.message)
  
  # Introspect handler's keyword parameters
  kwargs = {}
  handler.parameters.each do |(type, name)|
    next unless type == :key || type == :keyreq  # keyword args only
    
    # Find managed state matching the parameter name
    @builder.managed_state.each do |state|
      if matches_param?(state, name)  # e.g., DbConnection -> :db
        kwargs[name] = state
        break
      end
    end
  end
  
  handler.call(invoke, **kwargs)
end

def matches_param?(state, param_name)
  # DbConnection → db, UserService → users
  state.class.name.gsub(/([a-z])([A-Z])/, '\1_\2').downcase.start_with?(param_name.to_s)
end
```
